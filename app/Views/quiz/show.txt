<script>
    (() => {
        const form = document.getElementById('quiz-player-form');
        const steps = Array.from(form.querySelectorAll('.quiz-step'));
        const currentNumber = document.getElementById('quiz-current-number');
        const progressBar = document.getElementById('quiz-progress-bar');
        const questionCount = <?= $questionCount ?>;
        let activeStep = 0;

        const showStep = (nextStep) => {
            activeStep = nextStep;
            steps.forEach((step, index) => { step.classList.toggle('is-active', index === activeStep); step.setAttribute('aria-hidden', String(index !== activeStep)); });
            const questionNumber = Math.max(1, activeStep);
            currentNumber.textContent = questionNumber;
            progressBar.style.width = `${(questionNumber / questionCount) * 100}%`;
            form.querySelector('.is-active input, .is-active button')?.focus();
        };

        form.addEventListener('click', (event) => {
            const nextButton = event.target.closest('[data-next]');
            const previousButton = event.target.closest('[data-previous]');
            if (previousButton) { showStep(activeStep - 1); return; }
            if (!nextButton) return;

            const step = steps[activeStep];
            const error = step.querySelector('.quiz-step-error');
            if (activeStep === 0 && !step.querySelector('#participant_name').value.trim()) { step.querySelector('#participant_name').focus(); return; }
            if (activeStep === 0 && !step.querySelector('#participant_age').value) { step.querySelector('#participant_age').focus(); return; }
            if (activeStep > 0 && !step.querySelector('input[type="radio"]:checked')) { error.hidden = false; return; }
            if (error) error.hidden = true;
            if (activeStep === questionCount) { form.submit(); return; }
            showStep(activeStep + 1);
        });

        form.addEventListener('change', (event) => {
            const option = event.target.closest('.quiz-option');
            if (option) {
                option.closest('.quiz-options').querySelectorAll('.quiz-option').forEach((item) => item.classList.toggle('is-selected', item === option));
            }
            event.target.closest('.quiz-step')?.querySelector('.quiz-step-error')?.setAttribute('hidden', 'hidden');
        });
        showStep(0);
    })();
</script>
